import glob
import json
import os
import shutil
import sqlite3

DB_NAME = "recipes.db"
BACKUP_NAME = "recipes_before_update.db"


def ensure_schema(cursor):
    """Создает таблицы при первом запуске и добавляет колонку footnotes в существующую БД."""
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS recipes (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            recipe_number INTEGER UNIQUE,
            section TEXT,
            title TEXT NOT NULL,
            footnotes TEXT,
            technology TEXT,
            quality_requirements TEXT
        );
    """)

    cursor.execute("""
        CREATE TABLE IF NOT EXISTS recipe_variants (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            recipe_id INTEGER NOT NULL,
            category TEXT NOT NULL,
            output_weight TEXT,
            FOREIGN KEY (recipe_id) REFERENCES recipes(id) ON DELETE CASCADE
        );
    """)

    cursor.execute("""
        CREATE TABLE IF NOT EXISTS ingredients (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            variant_id INTEGER NOT NULL,
            alternative_group INTEGER NOT NULL,
            name TEXT NOT NULL,
            gross_weight TEXT,
            net_weight REAL,
            is_subtotal INTEGER DEFAULT 0,
            FOREIGN KEY (variant_id) REFERENCES recipe_variants(id) ON DELETE CASCADE
        );
    """)

    # Проверяем наличие колонки footnotes в старой таблице recipes
    cursor.execute("PRAGMA table_info(recipes)")
    columns = [col[1] for col in cursor.fetchall()]
    if "footnotes" not in columns:
        cursor.execute("ALTER TABLE recipes ADD COLUMN footnotes TEXT")
        print("В таблицу recipes добавлена новая колонка: footnotes")


def sync_recipe(recipe, cursor):
    """Добавляет новый рецепт или обновляет существующий, сохраняя ID."""
    num = recipe.get("recipe_number")
    title = recipe.get("title")
    footnotes = recipe.get("footnotes")

    cursor.execute("SELECT id FROM recipes WHERE recipe_number = ?", (num,))
    row = cursor.fetchone()

    if row:
        recipe_id = row[0]
        cursor.execute(
            """
            UPDATE recipes
            SET section = ?, title = ?, footnotes = ?, technology = ?, quality_requirements = ?
            WHERE id = ?
            """,
            (
                recipe.get("section"),
                title,
                footnotes,
                recipe.get("technology"),
                recipe.get("quality_requirements"),
                recipe_id,
            ),
        )
        action = "Обновлен"
    else:
        cursor.execute(
            """
            INSERT INTO recipes (recipe_number, section, title, footnotes, technology, quality_requirements)
            VALUES (?, ?, ?, ?, ?, ?)
            """,
            (
                num,
                recipe.get("section"),
                title,
                footnotes,
                recipe.get("technology"),
                recipe.get("quality_requirements"),
            ),
        )
        recipe_id = cursor.lastrowid
        action = "Добавлен"

    incoming_categories = set()

    for variant in recipe.get("variants", []):
        cat = variant.get("category")
        raw_out = variant.get("output_weight")
        out_weight = str(raw_out) if raw_out is not None else None
        incoming_categories.add(cat)

        cursor.execute(
            "SELECT id FROM recipe_variants WHERE recipe_id = ? AND category = ?",
            (recipe_id, cat),
        )
        var_row = cursor.fetchone()

        if var_row:
            variant_id = var_row[0]
            cursor.execute(
                "UPDATE recipe_variants SET output_weight = ? WHERE id = ?",
                (out_weight, variant_id),
            )
            cursor.execute("DELETE FROM ingredients WHERE variant_id = ?", (variant_id,))
        else:
            cursor.execute(
                "INSERT INTO recipe_variants (recipe_id, category, output_weight) VALUES (?, ?, ?)",
                (recipe_id, cat, out_weight),
            )
            variant_id = cursor.lastrowid

        for ing in variant.get("ingredients", []):
            cursor.execute(
                """
                INSERT INTO ingredients (
                    variant_id, alternative_group, name, gross_weight, net_weight, is_subtotal
                )
                VALUES (?, ?, ?, ?, ?, ?)
                """,
                (
                    variant_id,
                    ing.get("alternative_group"),
                    ing.get("name"),
                    ing.get("gross_weight"),
                    ing.get("net_weight"),
                    1 if ing.get("is_subtotal") else 0,
                ),
            )

    # Удаляем устаревшие колонки категорий, если они исчезли после повторного парсинга
    if incoming_categories:
        placeholders = ",".join("?" for _ in incoming_categories)
        cursor.execute(
            f"DELETE FROM recipe_variants WHERE recipe_id = ? AND category NOT IN ({placeholders})",
            [recipe_id, *incoming_categories],
        )

    print(f"  -> {action}: Рецепт №{num} — {title}")


def main():
    if os.path.exists(DB_NAME):
        shutil.copyfile(DB_NAME, BACKUP_NAME)
        print(f"Создана резервная копия: {BACKUP_NAME}\n")

    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("PRAGMA foreign_keys = ON;")

    ensure_schema(cursor)

    json_files = sorted(glob.glob("*.json"))
    if not json_files:
        print("В папке не найдено ни одного .json файла!")
        conn.close()
        return

    for filepath in json_files:
        print(f"Обработка файла: {filepath}...")
        with open(filepath, "r", encoding="utf-8") as file:
            try:
                data = json.load(file)
            except json.JSONDecodeError as e:
                print(f"  [ОШИБКА] Файл {filepath} содержит сломанный JSON: {e}")
                continue

        for recipe in data.get("recipes", []):
            sync_recipe(recipe, cursor)

    conn.commit()
    conn.close()
    print(f"\nГотово! База данных {DB_NAME} успешно обновлена.")


if __name__ == "__main__":
    main()
